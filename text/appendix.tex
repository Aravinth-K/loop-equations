\section{Analytic Structure and Asymptotics}
\label{app:asympt}

In this appendix we list the asymptotic expansions sheet by sheet, with
coefficients in the parameters of the corresponding potential, retaining
terms through $z^{-1}$. The physical
sheet is labelled $0$ and has $W(z)_0=z^{-1}+\mathcal O(z^{-2})$.
The remaining labels enumerate the local branches at infinity, keeping
ramified families together. We follow the presentation of the expansions
in \cite{Atkin:2015ksy}; the labels here do not prescribe the finite branch cuts.
The accompanying notebooks contain the complete calculations \cite{github:ak}.

\subsection{3-state Potts Model}

\subsubsection{Fixed boundary conditions}

The asymptotic behaviour of the planar resolvent $W_{(1)}(z)$ is given by:
%
\begin{align*}
    W_{(1)}(z)_0 & =   \frac{1}{z} + \mathcal{O}(z^{-2}) \\
    W_{(1)}(z)_1 & =  t_3 z^2 + (t_2 -1)z - 2t_3^{-1/2} z^{1/2} + \frac{t_2 -2}{t_3} - \frac{(t_2-2)^2}{4t_3^{3/2}} z^{-1/2} + \mathcal{O}(z^{-3/2}) \\
    W_{(1)}(z)_2 & =  t_3 z^2 + (t_2 -1)z + 2t_3^{-1/2}z^{1/2} + \frac{t_2 -2}{t_3} + \frac{(t_2-2)^2}{4t_3^{3/2}} z^{-1/2} + \mathcal{O}(z^{-3/2}) \\
    W_{(1)}(z)_3 & =  t_3 z^2 + \left(t_2-\frac12\right) z - \frac{t_3^{-1/2}}{2\sqrt{2}} z^{1/2} + \frac{(10t_2-1)}{8t_3}  \\* & - \frac{(4t_2^2 - 20t_2 +1)} {32\sqrt{2}t_3^{3/2}}z^{-1/2} - \frac{1}{2} z^{-1} + \mathcal{O}(z^{-3/2}) \\
    W_{(1)}(z)_4 & =  t_3 z^2 + \left(t_2-\frac12\right) z + \frac{t_3^{-1/2}}{2\sqrt{2}} z^{1/2} + \frac{(10t_2-1)}{8t_3} \\* & + \frac{(4t_2^2 - 20t_2 +1)}{32\sqrt{2}t_3^{3/2}}z^{-1/2} - \frac{1}{2} z^{-1} + \mathcal{O}(z^{-3/2})
\end{align*}

\subsubsection{Mixed boundary conditions}

The asymptotic behaviour of the planar resolvent $W_{(2)}(z)$ is given by:
%
\begin{align*}
    W_{(2)}(z)_0 & =   z^{-1} + \mathcal{O}(z^{-2}) \\
    W_{(2)}(z)_1 & = \frac{t_3 z^2}{4}+\frac{\left(t_2 - 2\right)}{2}z -t_3^{-1/2}z^{1/2} +\frac{t_2-1}{2 t_3}-\frac{(t_2-1)^2}{8 t_3^{3/2}}z^{-1/2}+\frac{1}{2}z^{-1} + \mathcal{O}(z^{-3/2})  \\
    W_{(2)}(z)_2 & = \frac{t_3 z^2}{4}+\frac{\left(t_2 - 2\right)}{2}z+t_3^{-1/2}z^{1/2}+\frac{t_2-1}{2 t_3}+\frac{(t_2-1)^2}{8 t_3^{3/2}}z^{-1/2}+\frac{1}{2}z^{-1}+ \mathcal{O}(z^{-3/2}) \\
    W_{(2)}(z)_3 & = -2 z + (1+i)t_3^{-1/2}z^{1/2} -\frac{2 t_2}{t_3}  +\frac{\left(1-i\right) \left(t_2^2-(6-4 i) t_2+1\right)}{8t_3^{3/2}}z^{-1/2} + \mathcal{O}(z^{-3/2})  \\
    W_{(2)}(z)_4 & = -2 z -(1-i)t_3^{-1/2}z^{1/2} -\frac{2 t_2}{t_3}  - \frac{\left(1+i\right) \left(t_2^2-(6+4 i) t_2+1\right)}{8t_3^{3/2}}z^{-1/2} + \mathcal{O}(z^{-3/2})  \\
    W_{(2)}(z)_5 & = -2 z -(1+i)t_3^{-1/2}z^{1/2} -\frac{2 t_2}{t_3}  -\frac{\left(1-i\right) \left(t_2^2-(6-4 i) t_2+1\right)}{8t_3^{3/2}}z^{-1/2}+ \mathcal{O}(z^{-3/2})  \\
    W_{(2)}(z)_6 & = -2 z + (1-i) t_3^{-1/2}z^{1/2} -\frac{2 t_2}{t_3}  +\frac{\left(1+i\right) \left(t_2^2-(6+4 i) t_2+1\right)}{8t_3^{3/2}}z^{-1/2} + \mathcal{O}(z^{-3/2}) \\
    W_{(2)}(z)_7 & = \frac{t_3 z^2}{4}+\frac{1}{2} \left(t_2-6\right) z+it_3^{-1/2}z^{1/2}-\frac{3 t_2+1}{2 t_3}-\frac{i \left(t_2^2-10 t_2+1\right)}{8 t_3^{3/2}}z^{-1/2} \\*
    & -\frac{1}{2}z^{-1} + \mathcal{O}(z^{-3/2}) \\
    W_{(2)}(z)_8 & = \frac{t_3 z^2}{4}+\frac{1}{2} \left(t_2-6\right) z-it_3^{-1/2}z^{1/2}-\frac{3 t_2+1}{2 t_3} +\frac{i \left(t_2^2-10 t_2+1\right)}{8 t_3^{3/2}}z^{-1/2} \\*
    & -\frac{1}{2}z^{-1}  +\mathcal{O}(z^{-3/2}) \\
    W_{(2)}(z)_9 & = -4z - \frac{4t_{2}}{t_3} - z^{-1} + \mathcal{O}(z^{-2})
\end{align*}

\subsubsection{Free boundary conditions}

The asymptotic behaviour of the planar resolvent $W_{(3)}(z)$ is given by:
%
\begin{align*}
    W_{(3)}(z)_0 & = z^{-1} + \mathcal{O}(z^{-2}) \\
    W_{(3)}(z)_1 & = \frac{t_3}{9} z^2 + \frac{(t_2-3)}{3} z + \frac{1}{z} + \mathcal{O}(z^{-2}) \\
    W_{(3)}(z)_2 & = -\frac{3}{2} z + \frac{3i}{2\sqrt{2}t_3^{1/2}}z^{1/2} - \frac{(18t_2 -9)}{8t_3} - \frac{3i(4t_2^2 - 36t_2 + 9)}{32\sqrt{2} t_3^{3/2}} z^{-1/2} \\*
    &\qquad{}- \frac{1}{2} z^{-1} + \mathcal{O}(z^{-3/2})  \\
    W_{(3)}(z)_3 & = -\frac{3}{2} z - \frac{3i}{2\sqrt{2}t_3^{1/2}}z^{1/2} - \frac{(18t_2 -9)}{8t_3} + \frac{3i(4t_2^2 - 36t_2 + 9)}{32\sqrt{2} t_3^{3/2}} z^{-1/2} \\*
    &\qquad{}- \frac{1}{2} z^{-1} + \mathcal{O}(z^{-3/2})  \\
    W_{(3)}(z)_4 & = - \frac{4}{3} z - \frac{2t_2}{t_3} - z^{-1} + \mathcal{O}(z^{-2})
\end{align*}
